nano frontend.py

from flask import Flask, jsonify
import requests
import random

app = Flask(__name__)

@app.route("/")
def home():
    return "Frontend Running"

@app.route("/order")
def order():

    amount = random.randint(100, 20000)

    payment = requests.get(
        f"http://localhost:5001/pay/{amount}"
    ).json()

    inventory = requests.get(
        "http://localhost:5002/check"
    ).json()

    return jsonify({
        "amount": amount,
        "payment": payment,
        "inventory": inventory
    })

if __name__ == "__main__":
    app.run(port=5000)
    
    
    
    
nano payment.py

from flask import Flask, jsonify
import random
import time

app = Flask(__name__)

@app.route("/")
def home():
    return "Payment Running"

@app.route("/pay/<int:amount>")
def pay(amount):

    transaction_id = random.randint(10000, 99999)

    time.sleep(1)

    if amount > 10000:
        status = "FLAGGED"
    else:
        status = "SUCCESS"

    return jsonify({
        "transaction_id": transaction_id,
        "amount": amount,
        "status": status
    })

if __name__ == "__main__":
    app.run(port=5001)
